\documentclass[tikz, border=10pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{patterns} % Librería necesaria para el patrón de rayas

\begin{document}
	\definecolor{maincolor1}{RGB}{1,82,142}
	\definecolor{maincolor2}{RGB}{21,182,184}
	\definecolor{mygrey}{rgb}{0.41, 0.41, 0.41}
	\definecolor{discardred}{RGB}{210, 60, 60}
	
	\begin{tikzpicture}
		\begin{axis}[
			title={$f(x) = x^2 - 3.2x + 3$ \quad (It. 3)},
			xlabel={$x$},
			ylabel={$f(x)$},
			xmin=-0.2, xmax=3.2,
			ymin=-0.1, ymax=3.3,
			axis lines=left,
			grid style={line width=.1pt, draw=gray!20},
			major grid style={line width=.2pt, draw=gray!50},
			minor tick num=1,
			legend pos=north west,
			legend style={font=\small, fill=white, fill opacity=0.9, draw opacity=1},
			width=12cm,
			height=8cm
			]
			
			% 1. Regiones descartadas con rayas diagonales rojas:
			% - [0.0, 1.4] descartada en la Iteración 1
			% - [1.95, 3.0] descartada tras la Iteración 3 (al ser f(1.75) < f(1.95))
			\addplot[
			draw=none, 
			pattern=north east lines, 
			pattern color=discardred
			] coordinates {
				(0.0, -0.1) (0.0, 3.3) (1.4, 3.3) (1.4, -0.1)
			};
			
			\addplot[
			draw=none, 
			pattern=north east lines, 
			pattern color=discardred
			] coordinates {
				(1.95, -0.1) (1.95, 3.3) (3.0, 3.3) (3.0, -0.1)
			};
			
			% 2. Nuevo intervalo de incertidumbre activo [1.4, 1.95] (Fondo gris)
			\addplot[
			draw=none, 
			fill=mygrey, 
			opacity=0.1
			] coordinates {
				(1.4, -0.1) (1.4, 3.3) (1.95, 3.3) (1.95, -0.1)
			};
			
			% 3. Curva continua f(x)
			\addplot [
			domain=0:3, 
			samples=100, 
			color=maincolor1, 
			thick
			] {x^2 - 3.2*x + 3};
			
			% 4. Línea del nuevo punto medio m2 = 1.85
			\draw[gray!60, dotted, line width=0.8pt] (axis cs:1.85, -0.1) -- (axis cs:1.85, 3.1);
			\node[above, font=\footnotesize, gray!80!black] at (axis cs:1.85, 3.1) {$m_2 = 1.85$};
			
			% 5. Líneas proyectadas para los nuevos puntos de prueba x1 = 1.75 y x2 = 1.95
			\draw[gray!50, dashed] (axis cs:1.75, -0.1) -- (axis cs:1.75, 0.4625);
			\draw[gray!50, dashed] (axis cs:1.95, -0.1) -- (axis cs:1.95, 0.5625);
			
			% 6. Puntos de prueba evaluados x1 y x2 en la Iteración 3
			\addplot [
			only marks, 
			mark=*, 
			mark size=3pt, 
			color=maincolor2
			] coordinates {
				(1.75, 0.4625)
				(1.95, 0.5625)
			};
			
			% 7. Mínimo analítico exacto (1.6, 0.44)
			\addplot [
			only marks, 
			mark=diamond*, 
			mark size=4pt, 
			color=maincolor2,
			thick
			] coordinates {(1.6, 0.44)};
			
			% Etiquetas en el eje X para los puntos x1 y x2
			\node[below, font=\footnotesize, maincolor1] at (axis cs:1.75, -0.1) {$x_1$};
			\node[below, font=\footnotesize, maincolor1] at (axis cs:1.95, -0.1) {$x_2$};
			
		\end{axis}
	\end{tikzpicture}
\end{document}
